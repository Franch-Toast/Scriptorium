# realtime_mutex
add_library(realtime_mutex ${LIBRARY_TYPE} mutex.cc mutex.h)
target_include_directories(realtime_mutex PUBLIC ${REPO_ROOT_PATH}/)

if(${USE_STATIC_LIB})
else()
  install(
    TARGETS realtime_mutex
    ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
    LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
    RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
endif()

if(BUILD_COMMON_MODULE_TEST)
    add_executable(mutex_test mutex_test.cc)
    target_link_libraries(mutex_test realtime_mutex ${GTEST_BOTH_LIBRARIES})
    add_test(mutex_test ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/mutex_test)
endif()
