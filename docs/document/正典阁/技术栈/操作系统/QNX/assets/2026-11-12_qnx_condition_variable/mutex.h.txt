#pragma once

#include <pthread.h>
#include <mutex>

#include <system_error>

namespace deeproute::base {

// Mutex with priority inheritance.
class RealtimeMutex {
 public:
  RealtimeMutex();
  RealtimeMutex(const RealtimeMutex&) = delete;
  RealtimeMutex& operator=(const RealtimeMutex&) = delete;
  inline ~RealtimeMutex() noexcept { pthread_mutex_destroy(&m_mutex); }

  inline void lock() {
    int result = pthread_mutex_lock(&m_mutex);
    if (result != 0) {
      throw std::system_error(result, std::system_category(),
                              "Failed to lock mutex");
    }
  }

  inline bool try_lock() noexcept {
    return pthread_mutex_trylock(&m_mutex) == 0;
  }

  inline void unlock() noexcept { pthread_mutex_unlock(&m_mutex); }

  inline pthread_mutex_t* native_handle() noexcept { return &m_mutex; }

 private:
  pthread_mutex_t m_mutex;
};

// Recursive mutex with priority inheritance.
class RecursiveRealtimeMutex {
 public:
  RecursiveRealtimeMutex();
  RecursiveRealtimeMutex(const RecursiveRealtimeMutex&) = delete;
  RecursiveRealtimeMutex& operator=(const RecursiveRealtimeMutex&) = delete;
  inline ~RecursiveRealtimeMutex() noexcept { pthread_mutex_destroy(&m_mutex); }

  inline void lock() {
    int result = pthread_mutex_lock(&m_mutex);
    if (result != 0) {
      throw std::system_error(result, std::system_category(),
                              "Failed to lock mutex");
    }
  }

  inline bool try_lock() noexcept {
    return pthread_mutex_trylock(&m_mutex) == 0;
  }

  inline void unlock() noexcept { pthread_mutex_unlock(&m_mutex); }

  inline pthread_mutex_t* native_handle() noexcept { return &m_mutex; }

 private:
  pthread_mutex_t m_mutex;
};

template <class MutexImp>
class LockGuard {
   typedef std::lock_guard< MutexImp > LockGuardImp;
   public:
    LockGuard(MutexImp& mutex) : lg_(mutex) {}
    ~LockGuard() {}
    LockGuardImp& native_handle() { return lg_; }
   private:
  LockGuardImp lg_;
};

template <class MutexImp>
class UniqueLock : public std::unique_lock<MutexImp> {
 public:
  UniqueLock(MutexImp& mutex) : std::unique_lock<MutexImp>(mutex) {}
  ~UniqueLock() {}
};

}  // namespace deeproute::base
